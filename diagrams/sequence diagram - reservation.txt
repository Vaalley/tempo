@startuml

title Diagramme de sequence — Réservation d'un espace (Tempo)

actor "Collaborateur" as User
participant "Frontend" as Frontend
participant "API" as API
database "PostgreSQL" as DB

== Consultation des espaces ==

User -> Frontend : Accéder à la page des espaces
activate Frontend

Frontend -> API : GET /api/workspaces
activate API

API -> DB : Récupérer tous les espaces
activate DB
DB --> API : Liste des espaces
deactivate DB

API --> Frontend : 200 OK
deactivate API

Frontend --> User : Afficher les espaces disponibles
deactivate Frontend

== Création de la réservation ==

User -> Frontend : Sélectionner un espace, choisir créneau, définir public/privé, saisir invités
activate Frontend

Frontend -> API : POST /api/bookings\n{ workspaceId, startAt, endAt, isPublic, invitedEmails[] }
activate API

API -> API : Vérifier le JWT

alt JWT invalide
    API --> Frontend : 401 Unauthorized
    deactivate API
    Frontend --> User : Afficher message d'erreur
    deactivate Frontend
else Authentification OK
end

API -> DB : Vérifier que l'espace existe
activate DB
DB --> API : Espace ou null
deactivate DB

alt Espace introuvable
    API --> Frontend : 404 Not Found
    deactivate API
    Frontend --> User : Afficher "Espace introuvable"
    deactivate Frontend
else Espace trouve
end

API -> DB : Vérifier les chevauchements de réservations
activate DB
DB --> API : Conflit ou OK
deactivate DB

alt Créneau déjà réservé
    API --> Frontend : 409 Conflict
    deactivate API
    Frontend --> User : Afficher "Créneau non disponible"
    deactivate Frontend
else Creneau disponible
end

API -> DB : Insérer la réservation (status: CONFIRMED, is_public, owner_id)
activate DB
DB --> API : Réservation créée avec ID
deactivate DB

alt Invitations à envoyer
    loop Pour chaque email invité
        API -> DB : INSERT INTO booking_invitations\n{ booking_id, email, status: PENDING }
        activate DB
        DB --> API : Invitation enregistrée
        deactivate DB
    end
end

API --> Frontend : 201 Created\n{ reservation, invitationsCount }
deactivate API

Frontend --> User : Afficher la confirmation et les invitations envoyées
deactivate Frontend

== Réception des réponses (asynchrone) ==

User -> Frontend : Recevoir notification invitation
activate Frontend
Frontend --> User : Afficher invitation reçue
deactivate Frontend

alt Invité accepte
    User -> Frontend : Accepter l'invitation
    activate Frontend
    Frontend -> API : POST /api/invitations/:id/accept
    activate API
    API -> DB : UPDATE booking_invitations SET status = 'ACCEPTED'
    activate DB
    DB --> API : OK
    deactivate DB
    API --> Frontend : 200 OK
    deactivate API
    Frontend --> User : Afficher "Invitation acceptée"
    deactivate Frontend
else Invité décline
    User -> Frontend : Décliner l'invitation
    activate Frontend
    Frontend -> API : POST /api/invitations/:id/decline
    activate API
    API -> DB : UPDATE booking_invitations SET status = 'DECLINED'
    activate DB
    DB --> API : OK
    deactivate DB
    API --> Frontend : 200 OK
    deactivate API
    Frontend --> User : Afficher "Invitation déclinée"
    deactivate Frontend
end

@enduml
