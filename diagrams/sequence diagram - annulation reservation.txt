@startuml

title Diagramme de sequence - Annulation d'une reservation (Tempo)

actor "Collaborateur" as User
participant "Frontend" as Frontend
participant "API" as API
database "PostgreSQL" as DB

== Consultation des reservations ==

User -> Frontend : Acceder a "Mes reservations"
activate Frontend

Frontend -> API : GET /api/bookings/my
activate API

API -> DB : SELECT * FROM bookings WHERE user_id = ? ORDER BY start_at DESC
activate DB
DB --> API : Liste des reservations
deactivate DB

API --> Frontend : 200 OK
deactivate API

Frontend --> User : Afficher les reservations
deactivate Frontend

== Annulation d'une reservation ==

User -> Frontend : Selectionner une reservation
activate Frontend

Frontend -> API : DELETE /api/bookings/:id
activate API

API -> API : Verifier le JWT

alt JWT invalide
    API --> Frontend : 401 Unauthorized
    deactivate API
    Frontend --> User : Afficher "Authentification requise"
    deactivate Frontend
else Authentification OK
end

API -> DB : SELECT * FROM bookings WHERE id = ? AND user_id = ?
activate DB
DB --> API : Reservation ou null
deactivate DB

alt Reservation introuvable
    API --> Frontend : 404 Not Found
    deactivate API
    Frontend --> User : Afficher "Reservation introuvable"
    deactivate Frontend
else Reservation trouvee
end

API -> API : Verifier delai d'annulation (start_at - now() > 24h)

alt Delai insuffisant (< 24h avant debut)
    API --> Frontend : 400 Bad Request
    deactivate API
    Frontend --> User : Afficher "Annulation impossible (delai trop court)"
    deactivate Frontend
else Delai OK
end

API -> DB : UPDATE bookings SET status = 'CANCELLED', cancelled_at = now() WHERE id = ?
activate DB
DB --> API : Reservation annulee
deactivate DB

API -> DB : SELECT user_id FROM booking_participants\nWHERE booking_id = ? AND status = 'ACCEPTED'
activate DB
DB --> API : Liste des participants (createur + invites)
deactivate DB

API -> API : Envoyer notification a chaque participant

API --> Frontend : 200 OK
deactivate API

Frontend --> User : Afficher confirmation d'annulation et notification
deactivate Frontend

@enduml
