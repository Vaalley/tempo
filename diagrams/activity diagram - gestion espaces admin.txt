@startuml

title Diagramme d'activité — Gestion des espaces (Administrateur)

|Administrateur|
start

:Accéder à la gestion des espaces;

:Choisir une action;

if (Action ?) then (Créer un espace)

    :Renseigner nom, type et capacité;
    :Confirmer la création;

    |Système|

    :Vérifier le rôle ADMIN (JWT);

    if (Rôle valide ?) then (non)
        :Retourner 403 Forbidden;
        stop
    else (oui)
    endif

    :Valider les données (nom, type, capacité);

    if (Données valides ?) then (non)
        :Retourner 400 Bad Request;
        |Administrateur|
        :Afficher les erreurs de saisie;
        stop
    else (oui)
    endif

    :Enregistrer l'espace en base;

    |Administrateur|
    :Afficher l'espace créé;

else (Supprimer un espace)

    :Sélectionner un espace;
    :Confirmer la suppression;

    |Système|

    :Vérifier le rôle ADMIN (JWT);

    if (Rôle valide ?) then (non)
        :Retourner 403 Forbidden;
        stop
    else (oui)
    endif

    :Rechercher l'espace par ID;

    if (Espace trouvé ?) then (non)
        :Retourner 404 Not Found;
        stop
    else (oui)
    endif

    :Rechercher les réservations futures\nsur cet espace;

    if (Réservations futures ?) then (oui)
        :Notifier les utilisateurs concernés\n(annulation de leurs réservations);
    else (non)
    endif

    :Supprimer l'espace;
    :Annuler les réservations associées\n(status: CANCELLED);
    :Enregistrer l'action dans le journal d'audit\n(MongoDB);

    |Administrateur|
    :Afficher la confirmation de suppression;

endif

stop

@enduml
