@startuml

title Diagramme d'activite - Gestion des espaces (Administrateur)

|Administrateur|
start
:Acceder a la gestion des espaces;
:Choisir une action;

if (Action ?) then (Creer un espace)
    :Renseigner nom, type, capacite\net quota maximum de personnes;
    :Confirmer la creation;

    |Systeme|
    :Verifier le role ADMIN (JWT);
    if (Role valide ?) then (non)
        :Retourner 403 Forbidden;
        |Administrateur|
        :Afficher "Acces refuse";
        stop
    else (oui)
    endif

    |Systeme|
    :Valider les donnees\n(nom, type, capacite, quota);
    if (Donnees valides ?) then (non)
        :Retourner 400 Bad Request;
        |Administrateur|
        :Afficher les erreurs de saisie;
        stop
    else (oui)
    endif

    |Systeme|
    :Enregistrer l'espace en base;

    |Administrateur|
    :Afficher l'espace cree;
    stop

else (Supprimer un espace)
    :Selectionner un espace;
    :Confirmer la suppression;

    |Systeme|
    :Verifier le role ADMIN (JWT);
    if (Role valide ?) then (non)
        :Retourner 403 Forbidden;
        |Administrateur|
        :Afficher "Acces refuse";
        stop
    else (oui)
    endif

    |Systeme|
    :Rechercher l'espace par ID;
    if (Espace trouve ?) then (non)
        :Retourner 404 Not Found;
        |Administrateur|
        :Afficher "Espace introuvable";
        stop
    else (oui)
    endif

    |Systeme|
    :Rechercher les reservations futures\nsur cet espace;
    if (Reservations futures ?) then (oui)
        :Notifier les utilisateurs concernes;
    else (non)
    endif
    :Supprimer l'espace;
    :Annuler les reservations associees\n(status: CANCELLED);
    :Enregistrer l'action dans le journal\nd'audit (MongoDB);

    |Administrateur|
    :Afficher la confirmation de suppression;
    stop
endif

@enduml
