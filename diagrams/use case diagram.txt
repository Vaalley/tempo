@startuml

left to right direction

' Acteurs
actor "Collaborateur\n(User)" as User
actor "Administrateur\n(Admin)" as Admin
actor "Système" as System

' L'Admin hérite des droits du User
Admin --|> User

rectangle "Tempo - Gestion d'espaces flex-office" {

    ' --- Authentification ---
    package "Authentification" {
        usecase "S'inscrire" as UC_Register
        usecase "Se connecter" as UC_Login
        usecase "Se déconnecter" as UC_Logout
    }

    ' --- Espaces de travail ---
    package "Gestion des espaces" {
        usecase "Consulter les espaces\ndisponibles" as UC_ListWorkspaces
        usecase "Filtrer par type\n(bureau / salle / cabine)" as UC_FilterWorkspaces
        usecase "Vérifier la disponibilité\nd'un espace" as UC_CheckAvailability
        usecase "Créer un espace" as UC_CreateWorkspace
        usecase "Modifier un espace" as UC_EditWorkspace
        usecase "Supprimer un espace" as UC_DeleteWorkspace
    }

    ' --- Réservations ---
    package "Gestion des réservations" {
        usecase "Créer une réservation\n(publique ou privée)" as UC_CreateBooking
        usecase "Inviter des collaborateurs\nà une réservation" as UC_InviteToBooking
        usecase "Consulter ses réservations" as UC_ViewMyBookings
        usecase "Annuler sa réservation" as UC_CancelBooking
        usecase "Confirmer sa présence\n(Check-in QR Code)" as UC_CheckIn
        usecase "Consulter toutes\nles réservations" as UC_ViewAllBookings
        usecase "Annuler n'importe\nquelle réservation" as UC_AdminCancelBooking
        usecase "Détecter les chevauchements" as UC_OverlapCheck
    }

    ' --- Tableau de bord admin ---
    package "Supervision & Audit" {
        usecase "Consulter le tableau\nde bord d'occupation" as UC_Dashboard
        usecase "Gérer les utilisateurs" as UC_ManageUsers
        usecase "Consulter les logs\nd'audit" as UC_ViewAuditLogs
        usecase "Enregistrer une action\ndans les logs" as UC_WriteAuditLog
    }

}

' --- Relations User ---
User --> UC_Register
User --> UC_Login
User --> UC_Logout

User --> UC_ListWorkspaces
User --> UC_FilterWorkspaces
User --> UC_CheckAvailability

User --> UC_CreateBooking
User --> UC_InviteToBooking
User --> UC_ViewMyBookings
User --> UC_CancelBooking
User --> UC_CheckIn

' --- Relations Admin (fonctionnalités exclusives) ---
Admin --> UC_CreateWorkspace
Admin --> UC_EditWorkspace
Admin --> UC_DeleteWorkspace

Admin --> UC_ViewAllBookings
Admin --> UC_AdminCancelBooking

Admin --> UC_Dashboard
Admin --> UC_ManageUsers
Admin --> UC_ViewAuditLogs

' --- Relations Système ---
System --> UC_WriteAuditLog

' --- Inclusions & Extensions ---
UC_CreateBooking .> UC_CheckAvailability : <<include>>
UC_CreateBooking .> UC_OverlapCheck : <<include>>

UC_ListWorkspaces .> UC_FilterWorkspaces : <<extend>>

' --- Notes ---
note bottom of UC_OverlapCheck
  Vérifie qu'aucune réservation
  existante ne chevauche les
  créneaux demandés (startAt/endAt)
end note

note bottom of UC_CheckIn
  Scan QR code dans la salle
  pour valider la présence physique
end note

note bottom of UC_CreateBooking
  Réservation publique : visible par tous
  Réservation privée : invitation uniquement
end note

@enduml
